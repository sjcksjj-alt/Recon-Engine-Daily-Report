function doPost(e){
  var ss=SpreadsheetApp.getActiveSpreadsheet();
  var d=JSON.parse(e.postData.contents);
  var sh=ss.getSheetByName(d.tab)||ss.insertSheet(d.tab);
  // Add header row if sheet is new
  if(sh.getLastRow()===0){sh.appendRow(['Date','Core Receipt','Inv Received','Inv Dispatched']);}
  var rows=sh.getDataRange().getValues();
  var found=false;
  for(var i=1;i<rows.length;i++){
    // Compare as string — date stored as YYYY-MM-DD text
    if(String(rows[i][0])===String(d.date)){
      sh.getRange(i+1,1,1,4).setValues([[d.date,d.core,d.inv_r,d.inv_d]]);
      found=true;break;
    }
  }
  if(!found)sh.appendRow([d.date,d.core,d.inv_r,d.inv_d]);
  return ContentService.createTextOutput("ok");
}
function doGet(e){
  var ss=SpreadsheetApp.getActiveSpreadsheet();
  var result={H:{},P15:{}};
  ['H','P15'].forEach(function(tab){
    var sh=ss.getSheetByName(tab);
    if(!sh)return;
    var rows=sh.getDataRange().getValues();
    for(var i=1;i<rows.length;i++){
      var dateVal=String(rows[i][0]).trim();
      // Accept only YYYY-MM-DD format keys
      if(/^\d{4}-\d{2}-\d{2}$/.test(dateVal)){
        result[tab][dateVal]={core:Number(rows[i][1])||0,inv_r:Number(rows[i][2])||0,inv_d:Number(rows[i][3])||0,ts:new Date().toISOString(),log:[]};
      }
    }
  });
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}
